\section{Basic Set Commands}
Here are some basic commands I made for sets
\begin{itemize}
    \item \textbackslash R makes $\R$
    \item \textbackslash RR makes $\RR$
    \item \textbackslash RRR makes $\RRR$
    \item \textbackslash C makes $\C$
    \item \textbackslash Q makes $\Q$
    \item \textbackslash Z\{\} makes $\Z{}$
    \item \textbackslash Z\curlb{n} makes $\Z{n}$
\end{itemize}

\section{Basic Algebra Commands}
Some basic commands common in algebra
\begin{itemize}
    \item \textbackslash iso makes $\iso$ (isometry)
    \item \textbackslash norm makes $\norm$
    \item \textbackslash normeq makes $\normeq$
    \item \textbackslash Ker\curlb{X} makes $\Ker{X}$
    \item \textbackslash factor\curlb{G}\curlb{H} makes $\factor{G}{H}$
    \item \textbackslash map\curlb{$\phi$}\curlb{A}\curlb{B} makes $\map{\phi}{A}{B}$
    \item \textbackslash cyclic\curlb{3} makes $\cyclic{3}$
    \item \textbackslash basicseries\curlb{H} makes $\basicseries{H}$
\end{itemize}
